2-D DP has a state with two indices: a grid cell `(r, c)`, a pair of prefixes `(i, j)` of two strings, or `(item, capacity)`. The recipe from [18](./18-dp-1d.md) is unchanged — brute force → state → memo → table → compress — but the table is a grid, the fill order matters more, and space compression to one row is a common follow-up.

---

## 1. Concepts to learn

| Concept | Explanation |
|---|---|
| Grid paths | `dp[r][c]` from `dp[r-1][c]` and `dp[r][c-1]` (top and left) — fill row by row |
| Two sequences | `dp[i][j]` = answer for `a[:i]` and `b[:j]`; padded with an empty-prefix row/column 0 |
| Match / mismatch transitions | LCS: match → `dp[i-1][j-1] + 1`, else `max(dp[i-1][j], dp[i][j-1])`. Edit distance: match → diagonal, else `1 + min(insert, delete, replace)` |
| Knapsack family | `(item, capacity)`: 0/1 → iterate capacity **down** in 1-D form; unbounded → **up**. Counting combinations (518): coins outer loop, amounts inner |
| State machines | `dp[i][state]` for stock problems (hold / sold / cooldown — 309) |
| Target sum as subset sum | `sum(P) - sum(N) = target` ⇒ `sum(P) = (total + target) / 2` (494) |
| Squares in a grid | `dp[r][c] = 1 + min(top, left, diag)` for maximal square side (221) |
| Fill order | Every cell's dependencies must be computed first — draw the arrows |
| Space compression | Only the previous row needed → keep 1-D `prev`/`cur` (or a single row updated carefully) |
| Complexity | rows × cols × transition cost; typically O(m·n) time, O(m·n) → O(min(m, n)) space |
| Memo on two indices | `@cache def f(i, j)` — hashable ints, fine; recursion depth up to m + n (watch Python's limits for 1000 + 1000) |

## 3. Python implementation

```python
from functools import cache


def unique_paths(m: int, n: int) -> int:
    row = [1] * n                                   # first row: one way to reach each cell
    for _ in range(1, m):
        for c in range(1, n):
            row[c] += row[c - 1]                    # row[c] (from above) + row[c-1] (from left)
    return row[-1]


def min_path_sum(grid: list[list[int]]) -> int:
    R, C = len(grid), len(grid[0])
    dp = [[0] * C for _ in range(R)]                # NEVER [[0] * C] * R
    for r in range(R):
        for c in range(C):
            if r == 0 and c == 0:
                dp[r][c] = grid[0][0]
            elif r == 0:
                dp[r][c] = dp[r][c - 1] + grid[r][c]
            elif c == 0:
                dp[r][c] = dp[r - 1][c] + grid[r][c]
            else:
                dp[r][c] = min(dp[r - 1][c], dp[r][c - 1]) + grid[r][c]
    return dp[-1][-1]


def lcs_table(a: str, b: str) -> int:
    m, n = len(a), len(b)
    dp = [[0] * (n + 1) for _ in range(m + 1)]      # row/col 0 = empty prefix
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if a[i - 1] == b[j - 1]:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    return dp[m][n]


def lcs_memo(a: str, b: str) -> int:
    @cache
    def f(i: int, j: int) -> int:                   # LCS of a[i:] and b[j:]
        if i == len(a) or j == len(b):
            return 0
        if a[i] == b[j]:
            return 1 + f(i + 1, j + 1)
        return max(f(i + 1, j), f(i, j + 1))
    return f(0, 0)


def lcs_two_rows(a: str, b: str) -> int:
    if len(b) > len(a):
        a, b = b, a                                 # keep the row as short as possible
    prev = [0] * (len(b) + 1)
    for i in range(1, len(a) + 1):
        cur = [0] * (len(b) + 1)
        for j in range(1, len(b) + 1):
            cur[j] = prev[j - 1] + 1 if a[i - 1] == b[j - 1] else max(prev[j], cur[j - 1])
        prev = cur
    return prev[-1]


def change_combinations(amount: int, coins: list[int]) -> int:
    """LC 518: coins OUTER loop → each combination counted once (not each permutation)."""
    dp = [1] + [0] * amount
    for c in coins:
        for a in range(c, amount + 1):              # upward: unlimited copies of c
            dp[a] += dp[a - c]
    return dp[amount]


def maximal_square(matrix: list[list[str]]) -> int:
    R, C = len(matrix), len(matrix[0])
    dp = [[0] * (C + 1) for _ in range(R + 1)]
    side = 0
    for r in range(1, R + 1):
        for c in range(1, C + 1):
            if matrix[r - 1][c - 1] == "1":
                dp[r][c] = 1 + min(dp[r - 1][c], dp[r][c - 1], dp[r - 1][c - 1])
                side = max(side, dp[r][c])
    return side * side


assert unique_paths(3, 7) == 28 and unique_paths(1, 1) == 1
assert min_path_sum([[1, 3, 1], [1, 5, 1], [4, 2, 1]]) == 7
for f in (lcs_table, lcs_memo, lcs_two_rows):
    assert f("abcde", "ace") == 3 and f("abc", "def") == 0 and f("", "a") == 0
assert change_combinations(5, [1, 2, 5]) == 4 and change_combinations(3, [2]) == 0
assert maximal_square([list("10100"), list("10111"), list("11111"), list("10010")]) == 4
bad = [[0] * 2] * 2
bad[0][0] = 1
assert bad[1][0] == 1                               # the aliasing trap, demonstrated
print("2-D DP templates ok")
```

## 4. The same in Java (occasional reps)

**This week's Java rep (Week 18): 1143 Longest Common Subsequence.**

| Python | Java |
|---|---|
| `[[0] * (n + 1) for _ in range(m + 1)]` | `int[][] dp = new int[m + 1][n + 1];` (zero-filled, independent rows) |
| `a[i - 1] == b[j - 1]` | `a.charAt(i - 1) == b.charAt(j - 1)` (chars compare with `==`, Strings with `.equals`) |
| `@cache def f(i, j)` | `int[][] memo` filled with −1 |
| counts "mod 10⁹+7" | `long` / `% MOD` |

```java
class LCSRep {
    static int longestCommonSubsequence(String a, String b) {
        int m = a.length(), n = b.length();
        int[][] dp = new int[m + 1][n + 1];
        for (int i = 1; i <= m; i++)
            for (int j = 1; j <= n; j++)
                dp[i][j] = a.charAt(i - 1) == b.charAt(j - 1)
                        ? dp[i - 1][j - 1] + 1
                        : Math.max(dp[i - 1][j], dp[i][j - 1]);
        return dp[m][n];
    }

    public static void main(String[] args) {
        System.out.println(longestCommonSubsequence("abcde", "ace") + " " + longestCommonSubsequence("abc", "def")); // 3 0
    }
}
```

## 6. How to recognise the pattern

- **Two** strings/arrays compared ("common subsequence", "edit", "interleave", "distinct subsequences").
- A grid where you move right/down (or any DAG order) and count/optimise paths.
- A 1-D DP whose state needs a second dimension: capacity, remaining target, holding a stock or not, "used k transactions".
- Constraints like m, n ≤ 1000 → O(m·n) = 10⁶ cells, fine in Python if the inner loop is simple (≈ 1 s); m, n ≤ 10⁴ each → too slow, look for 1-D.
- "Number of ways … modulo 10⁹ + 7" → counting DP.

## 11. Mistakes beginners commonly make

**Python-specific**
- `[[0] * n] * m` → every row is the same list; a write to one row "changes all rows". Use a comprehension.
- Updating a single-row DP in the wrong direction (0/1 needs downward, unbounded upward).
- Tuple assignment for state machines: `hold, sold, rest = max(hold, rest - p), hold + p, max(rest, sold)` — all right-hand values are the **old** ones (that's what you want); sequential assignments would use new values.
- `@cache` recursion on a 200×200 grid (329) → depth up to 4·10⁴ → `RecursionError` (and on 3.12+ the C-level limit for `@cache`); tabulate or use topological order.
- 221: comparing cells with `== 1` when they're strings `"1"`.
- 494: `(total + target) / 2` produces a float; check parity and use `//`; also `abs(target) > total` → 0.
- O(m·n) with heavy inner work (slices, `max` over lists) can TLE in Python even at 10⁶ cells — keep the inner loop to plain index arithmetic.

**General**
- Off-by-one between string indices and padded table indices (`a[i - 1]` for `dp[i]`).
- Wrong base row/column (Edit Distance: `dp[i][0] = i`, not 0).
- 518: loop order swapped → counts permutations.
- 63: forgetting that an obstacle in the first row blocks everything to its right.

**Java-rep traps (1143):** `charAt` vs indexing; `int[][]` sizes `m + 1`, `n + 1`.

## Worked example — 72. Edit Distance

**Clarify.** Strings `word1`, `word2` (0 ≤ length ≤ 500, lowercase). Operations: insert, delete, replace one character, each cost 1. Return the minimum number of operations to turn `word1` into `word2`. Empty → the other's length.

**Brute force.** Recursively try all three operations at each mismatch → O(3^(m+n)). Hopeless for 500.

**Optimise.** State: `dp[i][j]` = min operations to turn `word1[:i]` into `word2[:j]`.
- Base: `dp[i][0] = i` (delete everything), `dp[0][j] = j` (insert everything).
- If `word1[i-1] == word2[j-1]`: `dp[i][j] = dp[i-1][j-1]` (no cost).
- Else `1 + min(dp[i-1][j]` (delete from word1), `dp[i][j-1]` (insert into word1), `dp[i-1][j-1]` (replace)`)`.
Answer `dp[m][n]`. 501 × 501 ≈ 2.5·10⁵ cells — fast even in Python.

**Code (Python).**

```python
def min_distance(word1: str, word2: str) -> int:
    m, n = len(word1), len(word2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if word1[i - 1] == word2[j - 1]:
                dp[i][j] = dp[i - 1][j - 1]
            else:
                dp[i][j] = 1 + min(dp[i - 1][j],       # delete word1[i-1]
                                   dp[i][j - 1],       # insert word2[j-1]
                                   dp[i - 1][j - 1])   # replace
    return dp[m][n]


def min_distance_two_rows(word1: str, word2: str) -> int:
    prev = list(range(len(word2) + 1))                 # row i-1
    for i in range(1, len(word1) + 1):
        cur = [i] + [0] * len(word2)
        for j in range(1, len(word2) + 1):
            if word1[i - 1] == word2[j - 1]:
                cur[j] = prev[j - 1]
            else:
                cur[j] = 1 + min(prev[j], cur[j - 1], prev[j - 1])
        prev = cur
    return prev[-1]


for f in (min_distance, min_distance_two_rows):
    assert f("horse", "ros") == 3             # horse → rorse → rose → ros
    assert f("intention", "execution") == 5
    assert f("", "abc") == 3 and f("abc", "") == 3 and f("", "") == 0
    assert f("same", "same") == 0
print("72 worked example passed")
```

**Test cases.** `horse → ros` = 3 · `intention → execution` = 5 · empty source → insert all · empty target → delete all · identical strings → 0.

**Complexity.** Time O(m·n). Space O(m·n), or O(n) with two rows (keep the shorter string as the row to get O(min(m, n))).
